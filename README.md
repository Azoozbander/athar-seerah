function normalize(text){
return String(text).toLocaleLowerCase('ar')
.replace(/[\u064B-\u065F\u0670]/g,'')
.replace(/[أإآ]/g,'ا')
.replace(/ة/g,'ه')
.replace(/ى/g,'ي')
.replace(/[ـ]/g,'')
.trim();
}
